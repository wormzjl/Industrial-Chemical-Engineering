        var m=FluidTestSupport.networkModel();
        try(var out=new PrintWriter(Files.newBufferedWriter(Path.of("tmp/cooling-tower-review/table.csv")))){
            out.println("induced,cells,area,fill,lift,loading,inlet,speed,outlet,evaporation,air,airT,airW");
            var towers=new ArrayList<>(List.of(CoolingTower.induced(4,2,1,1),CoolingTower.induced(6,2,1,1),CoolingTower.induced(6,4,1,1),CoolingTower.induced(8,6,1,1),CoolingTower.natural(40),CoolingTower.natural(60),CoolingTower.natural(80)));

                double water=loading*tower.cellArea()*tower.cells();
                var r=OldTowerTransfer.fill(m,tower,water,inlet,speed);
                out.printf(Locale.ROOT,"%d,%d,%.6f,%.6f,%.6f,%.4f,%.2f,%.2f,%.6f,%.8f,%.6f,%.4f,%.8f%n",tower.draft()==CoolingTower.Draft.INDUCED?1:0,tower.cells(),tower.cellArea(),tower.fillDepth(),Math.max(1,tower.height()-tower.distributorHeight()),loading,inlet,speed,r.outletTemperature(),r.evaporation()/tower.cells(),r.dryAirFlow()/tower.cells(),r.airTemperature(),r.airHumidity());
            }
        }
    }
}
